Previously, user projects generated by `create-app` contained significant pipeline infrastructure (`pipeline-entry.ts`, dual Vite configs, build scripts) that users shouldn't need to understand or maintain. ### 1. Development CLI (`manifold-dev`)

Exported by configurator package, handles all development complexity:

```typescript
// @manifold-studio/configurator/bin/manifold-dev.js
export async function startDevelopment(options = {}) {
  const userProjectPath = process.cwd();

  // 1. Discover user's model files
  const modelFiles = await discoverUserModels(userProjectPath);
  console.log(
    "🔍 Discovered models:",
    modelFiles.map((f) => f.id)
  );

  // 2. Generate pipeline entry in our temp directory
  const pipelineEntry = generatePipelineEntry(modelFiles);
  await writeFile("./temp/user-pipeline-entry.ts", pipelineEntry);

  // 3. Start pipeline compiler (our Vite process)
  const pipelineServer = await startPipelineCompiler({
    entry: "./temp/user-pipeline-entry.ts",
    userProjectPath,
    watchPaths: [
      `${userProjectPath}/main.{ts,js}`,
      `${userProjectPath}/components/**/*.{ts,js}`,
    ],
  });

  // 4. Start UI server (serves built-in HTML + configurator)
  const uiServer = await startUIServer({
    userProjectPath,
    pipelinePath: "/temp/pipeline.js",
    isDevelopment: await detectConfiguratorDevelopment(),
  });

  // 5. Set up file watching for model discovery
  setupModelFileWatcher(userProjectPath, () => {
    // Regenerate pipeline entry when models added/removed
    regeneratePipelineEntry();
  });

  console.log("🚀 Manifold Studio development server started");
  console.log(`📱 UI: http://localhost:${uiServer.port}`);
  console.log(`⚙️  Pipeline: http://localhost:${pipelineServer.port}`);
}

async function detectConfiguratorDevelopment(): Promise<boolean> {
  // Check if we're in a workspace with configurator source
  const configuratorSrc = path.resolve("./packages/configurator/src");
  const hasConfiguratorSource = fs.existsSync(configuratorSrc);

  // Check if package.json has configurator as local dependency
  try {
    const packageJson = JSON.parse(fs.readFileSync("./package.json", "utf-8"));
    const hasLocalConfigurator =
      packageJson.devDependencies?.["@manifold-studio/configurator"]?.includes(
        "file:"
      );
    return hasConfiguratorSource || hasLocalConfigurator;
  } catch {
    return hasConfiguratorSource;
  }
}
```

### 2. Built-in HTML Generation and Development Mode

**Architecture Decision**: Instead of "embedding HTML in CLI", we make the CLI serve the configurator as a library - exactly like the current V3 setup, but without requiring users to have any HTML/JS files.

**Key Insights**:

- ✅ The configurator is already designed as a library with no build step required
- ✅ Current V3 setup uses `startConfigurator()` function with perfect HMR
- ✅ CSS/JS work directly from source with Vite handling TypeScript compilation
- ✅ TweakPane integration and web components are complete UX differentiators

**Template Strategy**:

- **Location**: `packages/configurator/templates/` (outside src for clean separation)
- **Content**: Minimal HTML template + configurator initialization script
- **Serving**: CLI reads templates and serves via Vite with dynamic configuration

**Development Mode Detection**:

- **Flag**: `--configurator-dev-mode` enables source-based development
- **Without flag**: Uses published configurator package (when available)
- **With flag**: Uses source-based Vite aliases (like current V3 setup)

**Vite Configuration Strategy**:

- **Dynamic Config Generation**: CLI generates temporary `vite.config.js` in user project
- **Source Aliases**: When in dev mode, points to configurator source directory
- **Avoids Publishing**: No need to publish configurator library initially

The CLI serves a built-in HTML page that automatically initializes the configurator, eliminating the need for users to create any HTML or JavaScript files:

```typescript
async function startUIServer(options) {
  const { userProjectPath, pipelinePath, isDevelopment } = options;

  const server = await createServer({
    root: userProjectPath,
    plugins: [
      {
        name: "manifold-auto-init",
        configureServer(server) {
          // Serve our built-in HTML for the root route
          server.middlewares.use("/", (req, res, next) => {
            if (req.url === "/" || req.url === "/index.html") {
              res.setHeader("Content-Type", "text/html");
              res.end(generateAutoInitHTML({ pipelinePath, isDevelopment }));
              return;
            }
            next();
          });
        },
      },
    ],
    resolve: isDevelopment
      ? {
          // In development mode, use source imports for HMR
          alias: {
            "@manifold-studio/configurator": path.resolve(
              "./packages/configurator/src"
            ),
            "@manifold-studio/wrapper": path.resolve("./packages/wrapper/src"),
          },
        }
      : undefined,
  });

  return server;
}
```

### 3. Dynamic Pipeline Entry Generation

```typescript
function generatePipelineEntry(modelFiles: ModelFile[]): string {
  const imports = modelFiles
    .map((model, i) => `import * as model${i} from '${model.relativePath}';`)
    .join("\n");

  const definitions = modelFiles
    .map(
      (model, i) => `{
    id: '${model.id}',
    path: '${model.relativePath}',
    name: '${model.name}',
    module: model${i}
  }`
    )
    .join(",\n  ");

  return `// Auto-generated by @manifold-studio/configurator
// DO NOT EDIT - This file is regenerated when model files change

${imports}

const modelDefinitions = [
  ${definitions}
];

// Export pipeline interface
export const pipeline = {
  getAvailableModels() {
    return modelDefinitions.map(def => ({
      id: def.id,
      name: def.name,
      path: def.path
    }));
  },
  
  async generateModel(modelId: string, config: any = {}) {
    const modelDef = modelDefinitions.find(def => def.id === modelId);
    if (!modelDef) {
      throw new Error(\`Model not found: \${modelId}\`);
    }
    
    const result = await modelDef.module.default(config);
    return result;
  },
  
  getModelConfig(modelId: string) {
    const modelDef = modelDefinitions.find(def => def.id === modelId);
    if (!modelDef?.module.config) return {};
    return modelDef.module.config;
  }
};

// Generate manifest data
export const manifestData = {
  models: modelDefinitions.map(def => ({
    id: def.id,
    name: def.name,
    path: def.path,
    config: def.module.config || {}
  })),
  timestamp: ${Date.now()}
};
`;
}
```

### 3. Model Discovery Service

```typescript
interface ModelFile {
  id: string; // 'main', 'wheel', 'bracket'
  name: string; // 'Main', 'Wheel', 'Bracket'
  relativePath: string; // './main.ts', './components/wheel.ts'
  absolutePath: string; // Full file system path
}

async function discoverUserModels(projectPath: string): Promise<ModelFile[]> {
  const patterns = [
    path.join(projectPath, "main.{ts,js}"),
    path.join(projectPath, "components/**/*.{ts,js}"),
  ];

  const files = await glob(patterns);
  const models: ModelFile[] = [];

  for (const file of files) {
    const relativePath = "./" + path.relative(projectPath, file);
    const id = extractModelId(relativePath);
    const name = extractModelName(relativePath);

    // Validate it's actually a model file
    if (await isValidModelFile(file)) {
      models.push({
        id,
        name,
        relativePath,
        absolutePath: file,
      });
    }
  }

  return models.sort((a, b) => {
    // Main model first, then alphabetical
    if (a.id === "main") return -1;
    if (b.id === "main") return 1;
    return a.name.localeCompare(b.name);
  });
}

function extractModelId(filePath: string): string {
  // './main.ts' -> 'main'
  // './components/wheel.ts' -> 'wheel'
  const basename = path.basename(filePath, path.extname(filePath));
  return basename;
}

async function isValidModelFile(filePath: string): Promise<boolean> {
  try {
    const content = await readFile(filePath, "utf-8");
    // Check for export default function or export default
    return /export\s+default\s+(function|async\s+function|\w+)/.test(content);
  } catch {
    return false;
  }
}
```

### 4. File Watching and Hot Reload

```typescript
function setupModelFileWatcher(projectPath: string, onModelChange: () => void) {
  const watcher = chokidar.watch(
    [
      path.join(projectPath, "main.{ts,js}"),
      path.join(projectPath, "components/**/*.{ts,js}"),
    ],
    {
      ignoreInitial: true,
    }
  );

  watcher.on("add", (filePath) => {
    console.log("📁 New model file detected:", filePath);
    onModelChange();
  });

  watcher.on("unlink", (filePath) => {
    console.log("🗑️  Model file removed:", filePath);
    onModelChange();
  });

  return watcher;
}

async function regeneratePipelineEntry() {
  const userProjectPath = process.cwd();
  const modelFiles = await discoverUserModels(userProjectPath);
  const pipelineEntry = generatePipelineEntry(modelFiles);

  await writeFile("./temp/user-pipeline-entry.ts", pipelineEntry);
  console.log(
    "🔄 Pipeline entry regenerated with",
    modelFiles.length,
    "models"
  );
}
```
